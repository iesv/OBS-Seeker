cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)

include(compilerconfig)
include(defaults)
include(helpers)

find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets)

add_custom_target(OBS-Seeker-prepare-dist
  COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/clean_dist.cmake"
  VERBATIM)

add_library(OBS-Seeker-core STATIC src/seek-core.cpp src/seek-core.hpp src/i18n.cpp src/i18n.hpp src/seeker_locales.qrc)
add_dependencies(OBS-Seeker-core OBS-Seeker-prepare-dist)
set_target_properties(OBS-Seeker-core PROPERTIES AUTORCC ON POSITION_INDEPENDENT_CODE ON)
target_link_libraries(OBS-Seeker-core PUBLIC Qt6::Core)
target_include_directories(OBS-Seeker-core PUBLIC src)

add_library(OBS-Seeker-ui STATIC src/seek-dock.cpp src/seek-dock.hpp src/window-activation.cpp src/window-activation.hpp
  src/presentation-escape.cpp src/presentation-escape.hpp)
set_target_properties(OBS-Seeker-ui PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(OBS-Seeker-ui PUBLIC OBS-Seeker-core Qt6::Widgets)
target_compile_definitions(OBS-Seeker-ui PUBLIC OBS_SEEKER_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_sources(OBS-Seeker-ui PRIVATE src/presentation-launch.cpp src/presentation-launch.hpp src/presentation-escape-windows.cpp)
  target_link_libraries(OBS-Seeker-ui PUBLIC shell32 ole32 user32)
  add_executable(OBS-Seeker-present WIN32 src/presentation-windows.cpp)
  target_link_libraries(OBS-Seeker-present PRIVATE ole32 oleaut32 shell32 comsuppw)
  install(TARGETS OBS-Seeker-present RUNTIME DESTINATION OBS-Seeker/bin/64bit)
elseif(APPLE)
  enable_language(OBJCXX)
  target_sources(OBS-Seeker-ui PRIVATE src/window-activation-macos.mm src/presentation-launch-macos.cpp src/presentation-escape-macos.mm)
  target_link_libraries(OBS-Seeker-ui PUBLIC "-framework AppKit" "-framework ApplicationServices")
  add_executable(OBS-Seeker-present MACOSX_BUNDLE src/presentation-macos.mm src/presentation-macos.js)
  set_source_files_properties(src/presentation-macos.js PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(OBS-Seeker-present PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/macos/presentation-helper.plist.in"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER com.andy.SeekerPresentation
    XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/macos/presentation-helper.entitlements"
    XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "-")
  target_compile_options(OBS-Seeker-present PRIVATE -fobjc-arc)
  target_link_libraries(OBS-Seeker-present PRIVATE "-framework AppKit" "-framework OSAKit" "-framework CoreServices")
  install(TARGETS OBS-Seeker-present BUNDLE DESTINATION OBS-Seeker.plugin/Contents/Helpers)
endif()

add_library(${CMAKE_PROJECT_NAME} MODULE src/plugin-main.cpp)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS-Seeker-ui OBS::libobs OBS::obs-frontend-api)
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})

include(CTest)
if(BUILD_TESTING)
  add_test(NAME seek-presentation-helper COMMAND OBS-Seeker-present)
  set_tests_properties(seek-presentation-helper PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
  if(APPLE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME seek-macos-presentation COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/presentation-macos-test.py"
      "$<TARGET_BUNDLE_DIR:OBS-Seeker-present>" "${CMAKE_CURRENT_SOURCE_DIR}/src/presentation-macos.js")
    set_tests_properties(seek-macos-presentation PROPERTIES TIMEOUT 90)
    add_executable(OBS-Seeker-escape-macos-tests tests/presentation-escape-macos-test.mm src/presentation-escape.cpp)
    target_include_directories(OBS-Seeker-escape-macos-tests PRIVATE src)
    target_compile_options(OBS-Seeker-escape-macos-tests PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(OBS-Seeker-escape-macos-tests PRIVATE Qt6::Core "-framework AppKit" "-framework ApplicationServices")
    add_test(NAME seek-presentation-escape-macos COMMAND OBS-Seeker-escape-macos-tests)
    set_tests_properties(seek-presentation-escape-macos PROPERTIES TIMEOUT 30)
  endif()
  if(WIN32)
    add_executable(OBS-Seeker-escape-fixture WIN32 tests/presentation-escape-fixture.cpp)
    set_target_properties(OBS-Seeker-escape-fixture PROPERTIES OUTPUT_NAME POWERPNT)
    target_link_libraries(OBS-Seeker-escape-fixture PRIVATE shell32 user32)
    add_executable(OBS-Seeker-escape-tests tests/presentation-escape-test.cpp)
    target_link_libraries(OBS-Seeker-escape-tests PRIVATE OBS-Seeker-ui)
    add_test(NAME seek-presentation-escape COMMAND OBS-Seeker-escape-tests "$<TARGET_FILE:OBS-Seeker-escape-fixture>")
    set_tests_properties(seek-presentation-escape PROPERTIES TIMEOUT 30)
  endif()
  add_executable(OBS-Seeker-tests tests/seek-core-test.cpp)
  target_link_libraries(OBS-Seeker-tests PRIVATE OBS-Seeker-core)
  add_test(NAME seek-core COMMAND OBS-Seeker-tests)
  add_executable(OBS-Seeker-ui-smoke tests/ui-smoke.cpp)
  set_target_properties(OBS-Seeker-ui-smoke PROPERTIES AUTOMOC ON)
  target_link_libraries(OBS-Seeker-ui-smoke PRIVATE OBS-Seeker-ui)
  foreach(locale IN ITEMS en-US zh-CN zh-TW)
    add_test(NAME seek-ui-${locale} COMMAND OBS-Seeker-ui-smoke "${locale}")
    if(APPLE)
      set(test_platform cocoa)
    else()
      set(test_platform windows)
    endif()
    set_tests_properties(seek-ui-${locale} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${test_platform}" TIMEOUT 60)
  endforeach()
  add_test(NAME seek-stress COMMAND OBS-Seeker-ui-smoke en-US --stress)
  set_tests_properties(seek-stress PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${test_platform}" TIMEOUT 180)
endif()
